% ConTeXt style for making a modern 2-page tech resume with pandoc.
% Modern Tech Minimalist Theme - Calibrated for 2 dense 95% full pages

\startmode[*mkii]
  \enableregime[utf-8]  
  \setupcolors[state=start]
\stopmode
$if(mainlang)$
\mainlanguage[$mainlang$]
$endif$

\setupcolor[hex]
\definecolor[titlecolor]   [h=0F172A] % Slate-900
\definecolor[sectioncolor] [h=1E40AF] % Blue-800
\definecolor[accentcolor]  [h=0284C7] % Sky-600
\definecolor[rulecolor]    [h=38BDF8] % Sky-400
\definecolor[subtlecolor]  [h=64748B] % Slate-500
\definecolor[lightrule]    [h=CBD5E1] % Slate-300

% Enable hyperlinks
\setupinteraction[state=start, color=accentcolor]

\setuppapersize [$if(papersize)$$papersize$$else$A4$endif$][$if(papersize)$$papersize$$else$A4$endif$]
\setuplayout    [width=middle, height=middle,
                 backspace=14mm, cutspace=14mm,
                 topspace=10mm, bottomspace=10mm,
                 header=0mm, footer=0mm]

\setupbodyfont[9.7pt, helvetica]

\setupwhitespace[none]

\setupblackrules[width=18mm, height=2.5pt, color=rulecolor]

% H1: Main Title / Name (Centered)
\setuphead[section]      [style=\tfc\bf, color=titlecolor, align=middle, before={\blank[none]}, after={\blank[none]}]
% H2: Categories (Left-aligned with cyan bar)
\setuphead[subsection]   [style=\tfa\bf, color=sectioncolor, align=flushleft,
                          before={\blank[small]\leavevmode\blackrule\hskip 0.6em}, after={\blank[none]}]
% H3: Subsections
\setuphead[subsubsection][style=\tf\bf, color=titlecolor, align=flushleft, before={\blank[none]}, after={\blank[none]}]

$if(number-sections)$
$else$
\setuphead[section, subsection, subsubsection][number=no]
$endif$

\definedescription
  [description]
  [headstyle=bold, style=normal,
   location=hanging, width=28mm, distance=3mm, margin=0cm,
   before={\blank[medium]}, after={\blank[none]}]

\setupitemize[autointro, packed, joinedup]
\setupitemize[indentnext=no, margin=0cm, leftmargin=1.2em]

\setupfloat[figure][default={here,nonumber}]
\setupfloat[table][default={here,nonumber}]

\setuptables[textwidth=max, HL=none]

\setupthinrules[width=100%, color=lightrule, height=0.5pt, before={\blank[none]}, after={\blank[none]}]

\definedelimitedtext[blockquote]

$if(toc)$
\setupcombinedlist[content][list={$placelist$}]

$endif$
$for(header-includes)$
$header-includes$
$endfor$

\starttext
$if(title)$
\startalignment[center]
  \blank[none]
  {\tfc $title$}
$if(author)$
  \blank[none]
  {\tf $for(author)$$author$$sep$\crlf $endfor$}
$endif$
$if(date)$
  \blank[none]
  {\tf $date$}
$endif$
  \blank[none]
\stopalignment
$endif$
$for(include-before)$
$include-before$
$endfor$
$if(toc)$
\placecontent
$endif$

$body$

$for(include-after)$
$include-after$
$endfor$
\stoptext
